cmake_minimum_required(VERSION 3.22)
project(SparkVisuals LANGUAGES CXX)
include(CTest)
if(BUILD_TESTING)
 add_executable(SparkVisualsMarkersTest tests/markers_test.cpp)
 target_compile_features(SparkVisualsMarkersTest PRIVATE cxx_std_20)
 add_test(NAME SparkVisualsMarkersTest COMMAND SparkVisualsMarkersTest)
 add_executable(SparkVisualsSettingsTest tests/settings_test.cpp)
 target_compile_features(SparkVisualsSettingsTest PRIVATE cxx_std_20)
 add_test(NAME SparkVisualsSettingsTest COMMAND SparkVisualsSettingsTest)
endif()
if(ANDROID)
 include(FetchContent)
 FetchContent_Declare(preloader_android
  GIT_REPOSITORY https://github.com/LiteLDev/preloader-android.git
  GIT_TAG 0.2.2)
 FetchContent_MakeAvailable(preloader_android)
 add_library(SparkVisuals SHARED src/SparkVisualsMod.cpp)
 target_compile_features(SparkVisuals PRIVATE cxx_std_20)
 target_link_libraries(SparkVisuals PRIVATE preloader)
 set_target_properties(SparkVisuals PROPERTIES OUTPUT_NAME "SparkVisuals")
endif()
